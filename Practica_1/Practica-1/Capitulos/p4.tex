\chapter{Requisitos No Funcionales}

\section{Rendimiento y Escalabilidad}
\begin{itemize}
    \item \textbf{RNF1:} El sistema debe procesar una reserva en menos de 2 segundos. El tiempo de carga de las páginas no debe superar 1 segundo en condiciones normales de tráfico.
    \item \textbf{RNF2:} El sistema debe soportar hasta 10,000 reservas diarias sin degradar el rendimiento y permitir la ampliación de servidores en temporadas altas.
    \item \textbf{RNF3:} El sistema debe estar disponible el 99.9\% del tiempo, exceptuando mantenimientos programados.
\end{itemize}

\subsection{Seguridad y Protección de Datos}
\begin{itemize}
    \item \textbf{RNF4:} Se requiere autenticación de usuario con contraseña y opción de autenticación en dos pasos (2FA). Solo los administradores pueden gestionar actividades y acceder a reportes financieros.
    \item \textbf{RNF5:} Toda la información sensible se almacenará cifrada cumpliendo con el RGPD. Los datos de tarjetas de crédito no se almacenarán en el sistema.
    \item \textbf{RNF6:} Cada código QR generado será único e intransferible, válido solo para una entrada.
\end{itemize}

\subsection{Usabilidad y Experiencia de Usuario}
\begin{itemize}
    \item \textbf{RNF7:} La plataforma debe ser accesible desde móviles y ordenadores, permitiendo completar una reserva en un máximo de tres clics.
    \item \textbf{RNF8:} Cumplimiento con WCAG 2.1 para garantizar accesibilidad (modo alto contraste, lectores de pantalla, etc.).
    \item \textbf{RNF9:} La plataforma debe estar disponible en español e inglés, como mínimo.
\end{itemize}

\subsection{Mantenibilidad y Actualización}
\begin{itemize}
    \item \textbf{RNF10:} El sistema debe ser modular para facilitar actualizaciones sin afectar la funcionalidad principal.
    \item \textbf{RNF11:} Se debe registrar en logs los eventos críticos, accesibles solo para administradores.
    \item \textbf{RNF12:} Las actualizaciones deben programarse en horarios de baja demanda y notificarse con 48 horas de antelación.
\end{itemize}

\subsection{Integración con otras Aplicaciones y Utilidades}
\begin{itemize}
    \item \textbf{RNF13:} Se debe integrar con PayPal y Google Pay.
    \item \textbf{RNF14:} Las reservas deben poder añadirse automáticamente al calendario del cliente.
\end{itemize}
